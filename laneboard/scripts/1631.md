Read-only queries (crew runs them on the copy and cites the output in its crew-handtest comment):

```sql
-- Q1 SO419944 lines, brands, location
SELECT l.line_no, p.product_code, b.brand_name, l.qty, l.stock_location
FROM projects.sales_orders so
JOIN projects.sales_order_lines l ON l.project_sales_order_id = so.id
JOIN products p ON p.id = l.product_id LEFT JOIN brands b ON b.id = p.brand_id
WHERE so.autocount_doc_no = 'SO419944' ORDER BY l.line_no;
-- Q2 free stock of those items in BRW-BB's group (codes ending -BB) and the BRW pool
SELECT p.product_code, w.warehouse_code, s.quantity_available
FROM projects.sales_orders so
JOIN projects.sales_order_lines l ON l.project_sales_order_id = so.id
JOIN products p ON p.id = l.product_id
JOIN stock s ON s.product_id = p.id JOIN warehouses w ON w.id = s.warehouse_id
WHERE so.autocount_doc_no = 'SO419944' AND (w.warehouse_code LIKE '%-BB' OR w.warehouse_code = 'BRW')
  AND s.quantity_available > 0 ORDER BY p.product_code, w.warehouse_code;
-- Q3 history: must return no rows
SELECT 'inquiry', oi.id::text FROM projects.order_inquiries oi
JOIN projects.sales_orders so ON so.id = oi.project_sales_order_id WHERE so.autocount_doc_no = 'SO419944'
UNION ALL
SELECT 'decision', d.revision_no::text FROM projects.so_supply_decisions d
JOIN projects.sales_orders so ON so.id = d.project_sales_order_id WHERE so.autocount_doc_no = 'SO419944';
-- Q4 local brand flag
SELECT brand_name, flows_to_purchasing FROM brands WHERE brand_name ILIKE 'TP ENTERPRISE%';
-- Q5 SO323450 SRTWC8826-SC raised rows
SELECT r.item_code, r.verb, r.qty, r.stock_location, r.state
FROM projects.order_inquiry_rows r JOIN projects.order_inquiries oi ON oi.id = r.order_inquiry_id
JOIN projects.sales_orders so ON so.id = oi.project_sales_order_id
WHERE so.autocount_doc_no = 'SO323450' AND r.item_code LIKE 'SRTWC8826-SC%' ORDER BY r.created_at;
```
